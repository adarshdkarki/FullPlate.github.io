],
"Traditional Beverages": [
{ id: 12, name: "Nepali Milk Tea (Chhiya)", price: 1, description: "Himalayan black tea brewed with whole milk, fresh ginger, crushed cardamom, cloves, and cinnamon." },
{ id: 13, name: "Himalayan Herbal Infusion", price: 1, description: "Herbal tea made with wild rhododendron petals, lemongrass, and mint leaves." }
]
},
addToCart(item) {
// Construct customization labels dynamically based on options picked
let details = [];
if (item.options) details.push(item.selectedOption || item.options[0]);
if (item.variants) details.push(item.selectedVariant || item.variants[0]);
let configString = details.join(' / ');

// Look for an existing match in the cart with exact same options
let match = this.cart.find(c => c.id === item.id && c.customization === configString);
if (match) {
match.quantity++;
} else {
this.cart.push({
id: item.id,
name: item.name,
price: item.price,
quantity: 1,
customization: configString
});
}
},
updateQuantity(index, amount) {
this.cart[index].quantity += amount;
if (this.cart[index].quantity <= 0) {
this.cart.splice(index, 1);
}
},
cartCount() {
return this.cart.reduce((sum, i) => sum + i.quantity, 0);
},
cartTotal() {
return this.cart.reduce((sum, i) => sum + (i.price * i.quantity), 0);
}
}
}
